.globl pow inc_arr

.data
fail_message: .asciiz "%s test failed\n"
pow_string: .asciiz "pow"
inc_arr_string: .asciiz "inc_arr"

success_message: .asciiz "Tests passed.\n"
array:
    .word 1 2 3 4 5
exp_inc_array_result:
    .word 2 3 4 5 6

.text
main:
    # pow: should return 2 ** 7 = 128
    li a0, 2
    li a1, 7
    jal ra pow
    li t0, 128 # verifies that pow returned the right value
    beq a0, t0, next_test
    la a0, pow_string
    j failure

next_test:
    # inc_arr: increments "array" in place
    la a0, array
    li a1, 5
    jal ra inc_arr
    jal ra check_arr # Verifies inc_arr returned the right value
    # all tests pass, exit normally
    li a0, 4
    la a1, success_message
    ecall
    li a0, 10
    ecall

# Computes a0 to the power of a1.
# uint32_t pow(uint32_t a0, uint32_t a1) {
#     uint32_t s0 = 1;
#     while (a1 != 0) { s0 *= a0; a1 -= 1; }
#     return s0;
# }
pow:
    # BEGIN PROLOGUE
    addi sp, sp, -4         # NEW: make space to save s0 (callee-saved)
    sw   s0, 0(sp)          # NEW: save s0 because we will modify it
    # END PROLOGUE
    li s0, 1
pow_loop:
    beq a1, zero, pow_end
    mul s0, s0, a0
    addi a1, a1, -1
    j pow_loop
pow_end:
    mv a0, s0
    # BEGIN EPILOGUE
    lw   s0, 0(sp)          # NEW: restore s0 before returning
    addi sp, sp, 4          # NEW: pop stack frame
    # END EPILOGUE
    jr ra

# Increments the elements of an array in-place.
# a0 = &array, a1 = length
# Calls helper_fn(&array[i]) which does "*p = *p + 1".
inc_arr:
    # BEGIN PROLOGUE
    addi sp, sp, -12        # NEW: space for ra,s0,s1 (we use s0/s1 here and call another fn)
    sw   ra, 0(sp)          # NEW: save ra because we call helper_fn (jal overwrites ra)
    sw   s0, 4(sp)          # NEW: save s0 (callee-saved, we modify it)
    sw   s1, 8(sp)          # NEW: save s1 (callee-saved, we modify it)
    # END PROLOGUE
    mv s0, a0               # Copy start of array to saved register
    mv s1, a1               # Copy length of array to saved register
    li t0, 0                # Initialize counter to 0
inc_arr_loop:
    beq t0, s1, inc_arr_end
    slli t1, t0, 2          # Convert array index to byte offset
    add a0, s0, t1          # a0 = &array[i]
    # Prepare to call helper_fn
    #
    # NEW: preserve t0 (loop counter) across the call.
    # Reason: t-registers are caller-saved and helper_fn may clobber them.
    addi sp, sp, -4         # NEW
    sw   t0, 0(sp)          # NEW

    jal ra helper_fn

    lw   t0, 0(sp)          # NEW: restore loop counter
    addi sp, sp, 4          # NEW: pop temp save
    # Finished call for helper_fn
    addi t0, t0, 1          # Increment counter
    j inc_arr_loop
inc_arr_end:
    # BEGIN EPILOGUE
    lw   ra, 0(sp)          # NEW: restore ra
    lw   s0, 4(sp)          # NEW: restore s0
    lw   s1, 8(sp)          # NEW: restore s1
    addi sp, sp, 12         # NEW: pop stack frame
    # END EPILOGUE
    jr ra

# "*a0 = *a0 + 1"
# Uses s0 in starter code — since s0 is callee-saved, we must save/restore it.
helper_fn:
    # BEGIN PROLOGUE
    addi sp, sp, -4         # NEW: space for s0
    sw   s0, 0(sp)          # NEW: save s0 (callee-saved) because we write to it
    # END PROLOGUE
    lw t1, 0(a0)
    addi s0, t1, 1          # uses s0, so we had to preserve it
    sw s0, 0(a0)
    # BEGIN EPILOGUE
    lw   s0, 0(sp)          # NEW: restore s0
    addi sp, sp, 4          # NEW: pop stack frame
    # END EPILOGUE
    jr ra

# YOU CAN IGNORE EVERYTHING BELOW THIS COMMENT

# Checks the result of inc_arr, which should contain 2 3 4 5 6 after one call.
check_arr:
    la t0, exp_inc_array_result
    la t1, array
    addi t2, t1, 20 # Last element is 5*4 bytes off
check_arr_loop:
    beq t1, t2, check_arr_end
    lw t3, 0(t0)
    lw t4, 0(t1)
    beq t3, t4, continue
    la a0, inc_arr_string
    j failure
continue:
    addi t0, t0, 4
    addi t1, t1, 4
    j check_arr_loop
check_arr_end:
    jr ra

# prints a failure message, then terminates the program
failure:
    mv a3, a0 # load the name of the test that failed
    li a0, 4  # String print ecall
    la a1, fail_message
    ecall
    li a0, 10 # Exit ecall
    ecall
